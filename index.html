Family Sharing Migration only (run if you already have the v2 tables):

    
-- Run this block ONCE if you already ran v2 SQL setup

create table if not exists family_members (
  id uuid default gen_random_uuid() primary key,
  owner_id uuid not null references auth.users(id) on delete cascade,
  member_id uuid references auth.users(id) on delete cascade,
  invite_code text not null unique check (char_length(invite_code) = 6),
  created_at timestamptz default now(),
  claimed_at timestamptz,
  constraint one_code_per_owner unique (owner_id)
);

alter table family_members enable row level security;
drop policy if exists "family_members_policy" on family_members;
drop policy if exists "family_code_claim" on family_members;
create policy "family_members_policy" on family_members for all
  using (owner_id = auth.uid() or member_id = auth.uid());
create policy "family_code_claim" on family_members for select using (true);

create or replace function chirp_effective_owner()
returns uuid language sql stable security definer as $$
  select coalesce(
    (select owner_id from family_members where member_id = auth.uid() limit 1),
    auth.uid()
  );
$$;

drop policy if exists "parent_kids" on kids;
drop policy if exists "parent_tasks" on tasks;
drop policy if exists "parent_completions" on completions;
drop policy if exists "parent_history" on history;
drop policy if exists "parent_timers" on timer_sessions;

create policy "parent_kids" on kids for all
  using (parent_id = chirp_effective_owner()) with check (parent_id = chirp_effective_owner());
create policy "parent_tasks" on tasks for all
  using (parent_id = chirp_effective_owner()) with check (parent_id = chirp_effective_owner());
create policy "parent_completions" on completions for all
  using (parent_id = chirp_effective_owner()) with check (parent_id = chirp_effective_owner());
create policy "parent_history" on history for all
  using (parent_id = chirp_effective_owner()) with check (parent_id = chirp_effective_owner());
create policy "parent_timers" on timer_sessions for all
  using (parent_id = chirp_effective_owner()) with check (parent_id = chirp_effective_owner());

    

    

    
Device Tracking Migration (run if you already have v2/v3 tables):

    
-- Add device_type and pool_id to existing tables (safe to re-run)
alter table history add column if not exists device_type text
  check (device_type in ('games','tv','tablet','computer'));
alter table timer_sessions add column if not exists device_type text
  check (device_type in ('games','tv','tablet','computer')) default 'games';
alter table timer_sessions add column if not exists pool_id uuid;

    

    

    
Variable Task Migration (run if you already have v2/v3 tables):

    
-- Add variable task columns to tasks and completions tables
alter table tasks add column if not exists rate_per_unit integer default null;
alter table tasks add column if not exists unit_label text default null;
alter table completions add column if not exists variable_units integer default null;
alter table completions add column if not exists variable_minutes integer default null;

    

    

    
Task Type Migration (Standing vs Project — run if you already have v3 tables):

    
-- Add task_type column. Standing = recurring chore (1 approval/day, streak-eligible).
-- Project = one-time job (archives after approval).
alter table tasks add column if not exists task_type text
  check (task_type in ('standing','project')) default 'standing';

-- Backfill from existing schedule values
update tasks set task_type = 'project' where schedule = 'none' and task_type is null;
update tasks set task_type = 'standing' where (schedule is null or schedule != 'none') and task_type is null;
